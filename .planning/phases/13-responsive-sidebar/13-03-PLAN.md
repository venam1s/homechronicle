          )}
          style={{ zIndex: 39 }}
          onClick={() => setDrawerOpen(false)}
          aria-label="Close navigation"
        />
        <div className={clsx('hc-drawer', drawerOpen && 'hc-drawer--open')} aria-modal="true">
          <Sidebar
            collapsed={false}
            onToggle={undefined}
            inDrawer={true}
            activePage={activePage}
            onNavigate={(page) => { setDrawerOpen(false); navigate(page); }}
            hasNewVersion={hasNewVersion}
            onOpenChangelog={openChangelog}
            buildVersion={__BUILD_VERSION__}
            availableUpdate={availableUpdate}
            onDismissUpdate={dismissUpdate}
          />
        </div>
      </>
    )}

    {/* Main grid layout */}
    <div className="app-layout" style={{ gridTemplateColumns: gridCols }}>
      {!isMobile && (
        <Sidebar
          collapsed={effectiveCollapsed}
          onToggle={isTablet ? undefined : () => setCollapsed(c => !c)}
          inDrawer={isTablet}
          activePage={activePage}
          onNavigate={navigate}
          hasNewVersion={hasNewVersion}
          onOpenChangelog={openChangelog}
          buildVersion={__BUILD_VERSION__}
          availableUpdate={availableUpdate}
          onDismissUpdate={dismissUpdate}
        />
      )}
      <main className={clsx('overflow-y-auto', isMobile && 'pt-12')}>
        {selectedAccessoryId ? (
          <ErrorBoundary key="detail"><AccessoryDetail accessoryId={selectedAccessoryId} onBack={handleAccessoryBack} /></ErrorBoundary>
        ) : (
          <ErrorBoundary key={activePage}>

        )}
      </main>

    {showChangelog && <ChangelogModal onClose={closeChangelog} />}
  </>
);

Key implementation notes:
- Mobile top bar uses `position: fixed` outside `.app-layout` grid -- avoids the 100dvh grid-row issue (Pitfall 2 from RESEARCH.md)
- `<main>` gets `pt-12` (48px) on mobile to clear the fixed top bar
- Drawer is always-mounted when `isMobile` -- CSS class toggle drives open/close animation (Pitfall 1 from RESEARCH.md)
- `effectiveCollapsed` is derived, never mutates `collapsed` -- preserves user preference across breakpoints (Pitfall 4)
- Backdrop uses inline `style={{ zIndex: 39 }}` because z-39 is not a standard Tailwind utility
- Desktop sidebar in tablet zone gets `inDrawer={isTablet}` (belt-and-suspenders to hide edge handle) and `onToggle={undefined}` (primary guard)
