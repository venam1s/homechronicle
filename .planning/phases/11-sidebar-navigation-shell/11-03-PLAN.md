    return (
      <>
        <div
          className="app-layout"
          style={{ gridTemplateColumns: collapsed ? '56px 1fr' : '240px 1fr' }}
        >
          <Sidebar
            collapsed={collapsed}
            onToggle={() => setCollapsed(c => !c)}
            activePage={activePage}
            onNavigate={(page) => {
              if (activePage === 'timeline' && page !== 'timeline') setTimelineInitialFilters(null);
              setSelectedAccessoryId(null);
              setActivePage(page);
            }}
            hasNewVersion={hasNewVersion}
            onOpenChangelog={handleOpenChangelog}

            availableUpdate={availableUpdate}
            onDismissUpdate={dismissUpdate}

              <ErrorBoundary key="accessory-detail">

              </ErrorBoundary>
            ) : (
              <ErrorBoundary key={activePage}>
                {activePage === 'dashboard' && (

                    dashboardDays={dashboardDays}
                    setDashboardDays={setDashboardDays}
                    syncDashboardRanges={syncDashboardRanges}
                    setSyncDashboardRanges={setSyncDashboardRanges}

                    onSelectRoom={handleSelectRoom}
                    onNavigateToAccessory={handleToastNavigateToAccessory}
                    onViewDeviceRepair={() => setActivePage('devices')}
                  />

                    onSelectAccessory={(id) => { handleSelectAccessory(id); }}
                  />
                )}
                {activePage === 'alerts' && <AlertsPage />}
                {activePage === 'settings' && <SettingsPage />}
              </ErrorBoundary>
            )}
          </main>
        </div>
        {showChangelog && <ChangelogModal onClose={handleCloseChangelog} />}
      </>
    );
